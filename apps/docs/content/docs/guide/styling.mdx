Masonix only handles layout. Your app owns the card UI, typography, imagery,
spacing, hover states, and semantic elements.

## Style the container

Use `className` and `style` on the outer Masonix container.

```tsx
<Masonry
  items={products}
  columnWidth={180}
  maxColumns={3}
  gap={16}
  className={cn('max-w-6xl', 'mx-auto px-4')}
  render={MasonryCard}
/>;

function MasonryCard({
  data,
}: {
  data: React.ComponentProps<typeof ProductCard>['product'];
}) {
  return <ProductCard product={data} />;
}

function cn(...groups: string[]) {
  return groups.join(' ');
}
```

## Style item wrappers

Use `itemClassName` when every item wrapper needs the same treatment.

```tsx
<MasonryBalanced
  items={articles}
  columns={3}
  gap={16}
  itemClassName={cn('overflow-hidden', 'rounded-xl')}
  render={MasonryCard}
/>;

function MasonryCard({
  data,
}: {
  data: React.ComponentProps<typeof ArticleCard>['article'];
}) {
  return <ArticleCard article={data} />;
}

function cn(...groups: string[]) {
  return groups.join(' ');
}
```

For `Masonry`, `columnClassName` styles the internal column wrappers in CSS
column mode.

## Card composition

Your render component receives `data`, `index`, and `width`.

```tsx
import type { MasonryRenderProps } from 'masonix';

function ProductCard({ data, width }: MasonryRenderProps<Product>) {
  return (
    <article className={cn('overflow-hidden', 'rounded-xl border', 'bg-white')}>
      <div
        className="relative"
        style={{ height: data.height, background: data.gradient }}
      >
        <span
          className={cn(
            'absolute left-3 top-3',
            'px-2 py-1',
            'rounded-full',
            'text-xs font-medium',
            'bg-white/85 text-zinc-900',
          )}
        >
          {data.badge}
        </span>
      </div>
      <div className="p-4">
        <p className={cn('text-xs', 'text-zinc-500')}>{data.category}</p>
        <h3 className={cn('mt-1', 'text-sm font-semibold')}>{data.name}</h3>
        <p className={cn('mt-3', 'text-xs', 'text-zinc-500')}>
          {width}px column · {data.price}
        </p>
      </div>
    </article>
  );
}

function cn(...groups: string[]) {
  return groups.join(' ');
}
```

Use `width` when media or canvas content needs to adapt to the actual column
width.

## Semantic markup

Use `as`, `itemAs`, `role`, and `aria-label` to make the layout fit your page
semantics.

```tsx
<Masonry
  as="ul"
  itemAs="li"
  role="list"
  aria-label="Featured products"
  className={cn('m-0 p-0', 'list-none')}
  itemClassName={cn('m-0 p-0', 'list-none')}
  items={products}
  render={MasonryCard}
/>;

function MasonryCard({
  data,
}: {
  data: React.ComponentProps<typeof ProductCard>['product'];
}) {
  return <ProductCard product={data} />;
}

function cn(...groups: string[]) {
  return groups.join(' ');
}
```
